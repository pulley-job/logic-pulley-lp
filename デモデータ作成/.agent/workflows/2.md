/**
 * デモデータ生成用GASテンプレート
 * エージェントはこの構成をベースに、ユーザーの要件に合わせて
 * generateDemoData()の中身とヘッダー定義を書き換えてください。
 */
function main() {
  // 1. スプレッドシート作成
  var fileName = "DemoData_" + Utilities.formatDate(new Date(), "JST", "yyyyMMdd_HHmm");
  var ss = SpreadsheetApp.create(fileName);
  var sheet = ss.getSheets()[0];
  sheet.setName("Database");

  // 2. カラム定義（ここを要件に合わせて可変にする）
  var headers = ["日付", "ID", "カテゴリ", "サブカテゴリ", "数値データ1", "数値データ2", "ステータス"];
  sheet.appendRow(headers);

  // 3. デモデータ生成（ここを要件に合わせて可変にする）
  var data = [];
  var rows = 100; // データ数
  
  for (var i = 0; i < rows; i++) {
    var date = new Date();
    date.setDate(date.getDate() - Math.floor(Math.random() * 365)); // 過去1年
    
    var id = "User_" + ("000" + (i + 1)).slice(-3);
    var category = ["A", "B", "C"][Math.floor(Math.random() * 3)];
    var subCategory = ["Type_X", "Type_Y"][Math.floor(Math.random() * 2)];
    var val1 = Math.floor(Math.random() * 1000) + 100;
    var val2 = Math.floor(Math.random() * 50);
    var status = ["完了", "進行中", "未着手"][Math.floor(Math.random() * 3)];

    data.push([date, id, category, subCategory, val1, val2, status]);
  }

  // 4. データ書き込み
  if (data.length > 0) {
    sheet.getRange(2, 1, data.length, data[0].length).setValues(data);
  }
  
  // 5. URL出力
  console.log("Spreadsheet Created: " + ss.getUrl());
}
